# Example domain configuration for Media Downloader.
#
# The extension creates domain-config.txt automatically on first use in
# ~/.config/media-downloader/ (or the "Domain Config Directory" preference).
# Run "Configure Domains" to open it.
#
# Format (one rule per line, all keys optional):
#   domain1, domain2: key="value" key='value' ...
#
# Keys:
#   yt-dlp        yt-dlp format selector for video, e.g. "bv*[height<=1080]+ba/b"
#   format        video container: mp4, mkv, webm, mov, avi, flv
#   quality       maximum video height: 2160, 1440, 1080, 720, ... or best
#   ffmpeg        extra ffmpeg arguments used when merging/remuxing video
#   compat        editing compatibility: original, prefer (H.264/AAC streams) or reencode
#   audio-format  audio format for the Audio command: m4a, mp3, opus, ogg, flac, wav
#   audio-ffmpeg  extra ffmpeg arguments used when extracting audio
#   args          any extra yt-dlp arguments
#
# A domain also matches its subdomains ("youtube.com" covers music.youtube.com).
# Priority: keys typed after the URL > this file > extension preferences.

# YouTube: up to 1080p in MP4, H.264/AAC when YouTube offers it
youtube.com, youtu.be: quality="1080" format="mp4" compat="prefer"

# YouTube Music: MP3 for the Audio command
music.youtube.com: audio-format="mp3"

# Instagram: always re-encode, its videos often have a variable frame rate
instagram.com: compat="reencode"

# Twitter/X
twitter.com, x.com: format="mp4"

# Vimeo: logged-in cookies from Firefox
vimeo.com: yt-dlp="bv*+ba/b" args="--cookies-from-browser firefox"

# TikTok: keep the video stream, convert the audio to AAC while merging
tiktok.com: format="mp4" ffmpeg="-c:v copy -c:a aac"

# Twitch VODs: keep the original MKV and skip the "prefer" sorting
twitch.tv: format="mkv" compat="original"
